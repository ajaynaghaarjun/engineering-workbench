cmake_minimum_required(VERSION 3.16)
project(EngineeringWorkbench VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(workbench_core src/core.cpp)
target_include_directories(workbench_core PUBLIC src)
target_link_libraries(workbench_core PUBLIC Threads::Threads)
add_executable(Launch src/main.cpp)
target_link_libraries(Launch PRIVATE workbench_core)
if(WIN32)
  target_link_libraries(Launch PRIVATE ws2_32 shell32)
  if(MINGW)
    target_link_options(Launch PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE workbench_core)
if(MINGW)
  target_link_options(core_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
add_custom_command(TARGET Launch POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/web $<TARGET_FILE_DIR:Launch>/web COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/data $<TARGET_FILE_DIR:Launch>/data)
